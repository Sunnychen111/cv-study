cmake_minimum_required(VERSION 3.16)

project(OpenCV_Demo)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(OpenCV REQUIRED)

add_executable(
    opencv_demo
    main.cpp
)

target_include_directories(
    opencv_demo
    PRIVATE
    ${OpenCV_INCLUDE_DIRS}
)

target_link_libraries(
    opencv_demo
    PRIVATE
    ${OpenCV_LIBS}
)
